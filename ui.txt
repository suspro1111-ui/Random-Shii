type Icon = {
    Url: string,
    Id: number,
    IconName: string,
    ImageRectOffset: Vector2,
    ImageRectSize: Vector2,
}

local FetchIcons, Icons = pcall(function()
    return (loadstring(
        game:HttpGet("https://raw.githubusercontent.com/mstudio45/lucide-roblox-direct/refs/heads/main/source.lua")
    ) :: () -> any)()
end)

local function GetCustomIcon(IconName: string): Icon?
    if not IconName then return nil end
    if tonumber(IconName) then
        IconName = string.format("rbxassetid://%s", tostring(IconName))
    end
    if typeof(IconName) == "string" and (IconName:match("^content://") or IconName:match("^rbxasset://") or IconName:match("^rbxassetid://") or IconName:match("roblox%.com/asset/%?id=")) then
        return {
            Url = IconName,
            Id = 0,
            IconName = IconName,
            ImageRectOffset = Vector2.zero,
            ImageRectSize = Vector2.zero,
        }
    end
    if FetchIcons and Icons then
        local Success, Icon = pcall(Icons.GetAsset, IconName)
        if Success and typeof(Icon) == "table" then return Icon end
    end
    return nil
end

function UIManager.SetIcon(imageLabel: ImageLabel, iconName: string)
    if not (imageLabel and imageLabel:IsA("ImageLabel")) then return end
    local icon = GetCustomIcon(iconName)
    if icon then
        imageLabel.Image = icon.Url or ""
        imageLabel.ImageRectOffset = icon.ImageRectOffset or Vector2.zero
        imageLabel.ImageRectSize = icon.ImageRectSize or Vector2.zero
    else
        imageLabel.Image = ""
        imageLabel.ImageRectOffset = Vector2.zero
        imageLabel.ImageRectSize = Vector2.zero
    end
end

local Acrylic = { Cleanups = {}, BlurInstances = {}, Folder = nil, DOF = nil, Enabled = UIManager.Config.ACRYLIC.BlurEnabled }

local function viewportPointToWorld(location, distance)
    local camera = Services.Workspace.CurrentCamera
    if not camera then return Vector3.zero end
    local unitRay = camera:ViewportPointToRay(location.X, location.Y)
    return unitRay.Origin + unitRay.Direction * distance
end

function Acrylic.Init()
    local camera = Services.Workspace.CurrentCamera
    if not camera then return end

    if Acrylic.Folder and Acrylic.Folder.Parent then Acrylic.Folder:Destroy() end
    local folder = Instance.new("Folder")
    folder.Name = "StandaloneAcrylicBlur"
    folder.Parent = camera
    Acrylic.Folder = folder

    pcall(function()
        local stale = Services.Lighting:FindFirstChild("StandaloneAcrylicDOF")
        if stale then stale:Destroy() end
    end)

    local dof = Instance.new("DepthOfFieldEffect")
    dof.Name = "StandaloneAcrylicDOF"
    dof.FarIntensity = 0
    dof.NearIntensity = 1
    dof.FocusDistance = UIManager.Config.ACRYLIC.FocusDistance or 2.0
    dof.InFocusRadius = UIManager.Config.ACRYLIC.InFocusRadius or 1.5
    dof.Enabled = Acrylic.Enabled
    dof.Parent = Services.Lighting
    Acrylic.DOF = dof

    table.insert(Acrylic.Cleanups, Services.Workspace:GetPropertyChangedSignal("CurrentCamera"):Connect(function()
        local newCam = Services.Workspace.CurrentCamera
        if newCam and Acrylic.Folder then Acrylic.Folder.Parent = newCam end
    end))
end

function Acrylic.SetEnabled(enabled)
    Acrylic.Enabled = enabled
    if Acrylic.DOF then Acrylic.DOF.Enabled = enabled end
    for _, blur in ipairs(Acrylic.BlurInstances) do
        if blur.SetVisibility then
            blur.SetVisibility(enabled and (blur.Frame and blur.Frame.Visible))
        end
    end
end

function Acrylic.CreateBlur(targetFrame, distance)
    local A = UIManager.Config.ACRYLIC
    distance = distance or A.Distance or 0.45
    local veilTransparency = A.VeilTransparency or 0.50

    local cleanups = {}
    local positions = { topLeft = Vector2.zero, topRight = Vector2.zero, bottomRight = Vector2.zero }

    local part = Instance.new("Part")
    part.Name = "AcrylicBody"
    part.Color = A.VeilColor or UIManager.Config.THEME.AcrylicMain
    part.Material = Enum.Material.Glass
    part.Size = Vector3.new(1, 1, 0)
    part.Anchored = true
    part.CanCollide = false
    part.Locked = true
    part.CastShadow = false
    part.Transparency = veilTransparency

    local mesh = Instance.new("SpecialMesh")
    mesh.MeshType = Enum.MeshType.Brick
    mesh.Offset = Vector3.new(0, 0, 0)
    mesh.Parent = part

    if Acrylic.Folder then part.Parent = Acrylic.Folder
    elseif Services.Workspace.CurrentCamera then part.Parent = Services.Workspace.CurrentCamera end

    local function render()
        local camera = Services.Workspace.CurrentCamera
        if not (camera and targetFrame and targetFrame.Parent) then return end
        local cframe = camera.CFrame or CFrame.new()

        local tl = viewportPointToWorld(positions.topLeft, distance)
        local tr = viewportPointToWorld(positions.topRight, distance)
        local br = viewportPointToWorld(positions.bottomRight, distance)

        local width = (tr - tl).Magnitude
        local height = (tr - br).Magnitude

        part.CFrame = CFrame.fromMatrix((tl + br) / 2, cframe.XVector, cframe.YVector, cframe.ZVector)
        mesh.Scale = Vector3.new(width, height, 0)
    end

    local function onChange()
        if not (targetFrame and targetFrame.Parent) then return end
        
        -- Exact inset calculation: prevents the 3D rectangle from sticking out of 2D rounded corners
        local cornerInset = 6
        local absPos = targetFrame.AbsolutePosition + Vector2.new(cornerInset, cornerInset)
        local absSize = targetFrame.AbsoluteSize - Vector2.new(cornerInset * 2, cornerInset * 2)

        positions.topLeft = absPos
        positions.topRight = absPos + Vector2.new(absSize.X, 0)
        positions.bottomRight = absPos + absSize
        task.spawn(render)
    end

    local camera = Services.Workspace.CurrentCamera
    if camera then
        table.insert(cleanups, camera:GetPropertyChangedSignal("CFrame"):Connect(render))
        table.insert(cleanups, camera:GetPropertyChangedSignal("ViewportSize"):Connect(function() onChange() render() end))
        table.insert(cleanups, camera:GetPropertyChangedSignal("FieldOfView"):Connect(render))
    end

    table.insert(cleanups, targetFrame:GetPropertyChangedSignal("AbsolutePosition"):Connect(onChange))
    table.insert(cleanups, targetFrame:GetPropertyChangedSignal("AbsoluteSize"):Connect(onChange))

    local blurObj = {
        Frame = targetFrame,
        Model = part,
        SetVisibility = function(visible)
            if part and part.Parent then
                part.Transparency = (visible and Acrylic.Enabled) and veilTransparency or 1
            end
        end,
        Destroy = function()
            for _, c in ipairs(cleanups) do pcall(function() c:Disconnect() end) end
            if part and part.Parent then part:Destroy() end
        end
    }

    targetFrame.Destroying:Connect(function() blurObj.Destroy() end)
    table.insert(Acrylic.BlurInstances, blurObj)
    onChange()
    return blurObj
end

function Acrylic.Destroy()
    for _, c in ipairs(Acrylic.Cleanups) do pcall(function() c:Disconnect() end) end
    for _, b in ipairs(Acrylic.BlurInstances) do pcall(function() b.Destroy() end) end
    table.clear(Acrylic.BlurInstances)
    if Acrylic.DOF then pcall(function() Acrylic.DOF:Destroy() end) end
    if Acrylic.Folder then pcall(function() Acrylic.Folder:Destroy() end) end
end

Acrylic.Init()
UIManager.Acrylic = Acrylic

function UIManager.MakeDraggable(frame, handle)
    handle = handle or frame
    local dragging = false
    local dragInput, dragStart, startPos

    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = frame.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)

    handle.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)

    Services.UserInputService.InputChanged:Connect(function(input)
        if input == dragInput and dragging then
            local delta = input.Position - dragStart
            frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
end

function UIManager.MakeResizable(frame, handle, minSize, maxSize)
    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            local startInput = input.Position
            local startPos = frame.Position
            local startSize = frame.Size
            local moveConn
            moveConn = Services.UserInputService.InputChanged:Connect(function(move)
                if move.UserInputType == Enum.UserInputType.MouseMovement or move.UserInputType == Enum.UserInputType.Touch then
                    local delta = move.Position - startInput
                    local newWidth  = math.clamp(startSize.X.Offset - delta.X, minSize.X, maxSize.X)
                    local newHeight = math.clamp(startSize.Y.Offset + delta.Y, minSize.Y, maxSize.Y)
                    frame.Size = UDim2.new(startSize.X.Scale, newWidth, startSize.Y.Scale, newHeight)
                    frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + (startSize.X.Offset - newWidth), startPos.Y.Scale, startPos.Y.Offset)
                end
            end)
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End and moveConn then
                    moveConn:Disconnect()
                    moveConn = nil
                end
            end)
        end
    end)
end

-- ---------------------------------------------------------------------
-- 6. LOGGING SYSTEM
-- ---------------------------------------------------------------------
function UIManager.Log(tag, text, colorHex, customTimestamp)
    local ok, err = pcall(function()
        if not UIManager.LogScroll then return end
        LogIndex = LogIndex + 1
        colorHex = colorHex or "#E6E6F0"
        
        local tagOut = (tostring(tag):gsub("[<>&]", { ["<"] = "&lt;", [">"] = "&gt;", ["&"] = "&amp;" }))
        local textOut = tostring(text)
        textOut = textOut:gsub("<=", "&lt;=")
        textOut = textOut:gsub(">=", "&gt;=")
        textOut = textOut:gsub("<<", "&lt;&lt;")
        textOut = textOut:gsub(">>", "&gt;&gt;")

        local timeStamp = customTimestamp or os.date("%H:%M:%S")
        local scroll = UIManager.LogScroll
        local nearBottom = (scroll.AbsoluteCanvasSize.Y - scroll.CanvasPosition.Y - scroll.AbsoluteWindowSize.Y) <= 48

        local entry = Instance.new("TextLabel")
        entry.Name = string.format("LogEntry_%d", LogIndex)
        entry.Size = UDim2.new(1, 0, 0, 0)
        entry.AutomaticSize = Enum.AutomaticSize.Y
        entry.BackgroundTransparency = 1
        entry.RichText = true
        entry.Font = UIManager.Config.UI_LAYOUT.LogFont
        entry.TextSize = UIManager.Config.UI_LAYOUT.LogTextSize
        entry.TextColor3 = UIManager.Config.THEME.Text
        entry.TextXAlignment = Enum.TextXAlignment.Left
        entry.TextWrapped = true
        entry.LayoutOrder = LogIndex
        entry.ZIndex = 6
        entry.Text = string.format("<font color='#584E6D'>[%s]</font> <font color='%s'><b>[%s]</b> %s</font>", timeStamp, colorHex, tagOut, textOut)
        entry.Parent = scroll

        local labels = {}
        for _, child in ipairs(scroll:GetChildren()) do
            if child:IsA("TextLabel") then table.insert(labels, child) end
        end
        table.sort(labels, function(a, b) return (a.LayoutOrder or 0) < (b.LayoutOrder or 0) end)
        while #labels > UIManager.Config.MAX_LOG_LINES do
            local oldest = table.remove(labels, 1)
            if oldest then pcall(function() oldest:Destroy() end) end
        end

        if nearBottom then
            task.defer(function()
                if entry.Parent then scroll.CanvasPosition = Vector2.new(0, scroll.AbsoluteCanvasSize.Y) end
            end)
        end
        entry.TextTransparency = 0.45
        Services.TweenService:Create(entry, TweenInfo.new(0.18, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), { TextTransparency = 0 }):Play()
    end)
    if not ok then warn("[Console] log error:", err) end
end

return UIManager
